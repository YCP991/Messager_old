# 麦思哲（MaiSiZhe）前端详细设计方案

## 1. 项目概述

### 1.1 项目名称
麦思哲(MaiSiZhe) - 智能Agent校园即时通讯系统前端

### 1.2 技术栈选型
- **核心框架**: Vue 3.4+ (Composition API)
- **语言**: TypeScript 5.x
- **构建工具**: Vite 5.x
- **状态管理**: Pinia 2.x
- **路由**: Vue Router 4.x
- **UI组件库**: Naive UI 2.x
- **HTTP客户端**: Axios 1.x
- **WebSocket**: 原生 WebSocket + @stomp/stompjs
- **Markdown渲染**: marked + highlight.js
- **工具库**: dayjs, lodash-es

### 1.3 设计目标
- 实现流畅的即时通讯体验(消息延迟 < 200ms)
- 支持WebSocket断线自动重连与消息补盲
- 提供优雅的AI交互界面(流式输出、思考动画)
- 实现响应式布局,支持桌面端与移动端基础适配
- 保证代码可维护性与团队协作效率

### 3.1 WebSocket管理器设计

#### 3.1.1 WebSocketManager类设计

```typescript
// utils/websocket.ts

type WSMessage = {
  type: string;
  data: any;
  clientMsgId?: string;
};

class WebSocketManager {
  private ws: WebSocket | null = null;
  private reconnectTimer: number | null = null;
  private heartbeatTimer: number | null = null;
  private messageQueue: WSMessage[] = [];
  private reconnectCount: number = 0;
  private maxReconnectCount: number = 10;
  private reconnectInterval: number = 3000;
  private heartbeatInterval: number = 25000;
  
  // 事件回调
  private onMessageCallback: ((data: any) => void) | null = null;
  private onConnectedCallback: (() => void) | null = null;
  private onDisconnectedCallback: (() => void) | null = null;
  
  constructor(private url: string, private token: string) {}
  
  /**
   * 连接WebSocket
   */
  connect(): Promise<void> {
    return new Promise((resolve, reject) => {
      try {
        this.ws = new WebSocket(`${this.url}?token=${this.token}`);
        
        this.ws.onopen = () => {
          console.log('WebSocket连接成功');
          this.reconnectCount = 0;
          this.startHeartbeat();
          this.flushMessageQueue();
          this.onConnectedCallback?.();
          resolve();
        };
        
        this.ws.onmessage = (event) => {
          const data = JSON.parse(event.data);
          this.handleMessage(data);
        };
        
        this.ws.onerror = (error) => {
          console.error('WebSocket错误:', error);
          reject(error);
        };
        
        this.ws.onclose = () => {
          console.log('WebSocket连接关闭');
          this.stopHeartbeat();
          this.onDisconnectedCallback?.();
          this.attemptReconnect();
        };
      } catch (error) {
        reject(error);
      }
    });
  }
  
  /**
   * 发送消息
   */
  send(message: WSMessage): void {
    if (!message.clientMsgId) {
      message.clientMsgId = this.generateClientMsgId();
    }
    
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify(message));
    } else {
      // 连接未建立,加入队列
      this.messageQueue.push(message);
    }
  }
  
  /**
   * 心跳机制
   */
  private startHeartbeat(): void {
    this.heartbeatTimer = window.setInterval(() => {
      this.send({ type: 'PING', data: null });
      
      // 5秒内未收到PONG则判定断开
      setTimeout(() => {
        if (this.ws) {
          this.ws.close();
        }
      }, 5000);
    }, this.heartbeatInterval);
  }
  
  private stopHeartbeat(): void {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = null;
    }
  }
  
  /**
   * 自动重连
   */
  private attemptReconnect(): void {
    if (this.reconnectCount >= this.maxReconnectCount) {
      console.error('达到最大重连次数');
      return;
    }
    
    this.reconnectCount++;
    const delay = this.reconnectInterval * Math.pow(2, this.reconnectCount - 1);
    
    console.log(`第${this.reconnectCount}次重连,延迟:${delay}ms`);
    
    this.reconnectTimer = window.setTimeout(() => {
      this.connect().catch(console.error);
    }, delay);
  }
  
  /**
   * 刷新消息队列
   */
  private flushMessageQueue(): void {
    while (this.messageQueue.length > 0) {
      const message = this.messageQueue.shift();
      if (message && this.ws) {
        this.ws.send(JSON.stringify(message));
      }
    }
  }
  
  /**
   * 生成客户端消息ID
   */
  private generateClientMsgId(): string {
    return `msg_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }
  
  /**
   * 断开连接
   */
  disconnect(): void {
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer);
    }
    this.stopHeartbeat();
    if (this.ws) {
      this.ws.close();
    }
  }
  
  // 设置回调
  onMessage(callback: (data: any) => void): void {
    this.onMessageCallback = callback;
  }
  
  onConnected(callback: () => void): void {
    this.onConnectedCallback = callback;
  }
  
  onDisconnected(callback: () => void): void {
    this.onDisconnectedCallback = callback;
  }
  
  private handleMessage(data: any): void {
    if (data.type === 'PONG') {
      return; // 心跳响应,不处理
    }
    this.onMessageCallback?.(data);
  }
}

export default WebSocketManager;
```

#### 3.1.2 WebSocket使用示例

```typescript
// composables/useWebSocket.ts

import { ref, onMounted, onUnmounted } from 'vue';
import WebSocketManager from '@/utils/websocket';
import { useUserStore } from '@/stores/user';
import { useChatStore } from '@/stores/chat';

export function useWebSocket() {
  const userStore = useUserStore();
  const chatStore = useChatStore();
  const wsManager = ref<WebSocketManager | null>(null);
  const isConnected = ref(false);
  
  onMounted(() => {
    initWebSocket();
  });
  
  onUnmounted(() => {
    wsManager.value?.disconnect();
  });
  
  function initWebSocket() {
    const token = userStore.token;
    const wsUrl = import.meta.env.VITE_WS_URL;
    
    wsManager.value = new WebSocketManager(wsUrl, token);
    
    wsManager.value.onConnected(() => {
      isConnected.value = true;
      console.log('WebSocket已连接');
    });
    
    wsManager.value.onDisconnected(() => {
      isConnected.value = false;
      console.log('WebSocket已断开');
    });
    
    wsManager.value.onMessage((data) => {
      handleWSMessage(data);
    });
    
    wsManager.value.connect().catch(console.error);
  }
  
  function handleWSMessage(data: any) {
    switch (data.type) {
      case 'PRIVATE_MESSAGE':
        chatStore.addPrivateMessage(data.data);
        break;
      case 'GROUP_MESSAGE':
        chatStore.addGroupMessage(data.data);
        break;
      case 'MESSAGE_RECALL':
        chatStore.recallMessage(data.data);
        break;
      case 'USER_ONLINE':
        userStore.updateOnlineStatus(data.data.userId, true);
        break;
      case 'USER_OFFLINE':
        userStore.updateOnlineStatus(data.data.userId, false);
        break;
      default:
        console.warn('未知消息类型:', data.type);
    }
  }
  
  function sendMessage(message: any) {
    wsManager.value?.send(message);
  }
  
  return {
    isConnected,
    sendMessage
  };
}
```

#### 3.2.1 Chat Store设计

```typescript
// stores/chat.ts

import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { Message, ChatSession } from '@/types/message';

export const useChatStore = defineStore('chat', () => {
  // 状态
  const activeChatId = ref<string>('');
  const messages = ref<Record<string, Message[]>>({});
  const lastSeqMap = ref<Record<string, number>>({});
  const sendQueue = ref<Message[]>([]);
  const chatSessions = ref<ChatSession[]>([]);
  
  // 计算属性
  const currentMessages = computed(() => {
    return messages.value[activeChatId.value] || [];
  });
  
  const unreadCount = computed(() => {
    return chatSessions.value.reduce((sum, session) => {
      return sum + (session.unreadCount || 0);
    }, 0);
  });
  
  // 方法
  function setActiveChat(chatId: string) {
    activeChatId.value = chatId;
  }
  
  function addMessage(chatId: string, message: Message) {
    if (!messages.value[chatId]) {
      messages.value[chatId] = [];
    }
    
    // 二分法插入排序,保证消息顺序
    const msgList = messages.value[chatId];
    const index = findInsertIndex(msgList, message.seqId);
    msgList.splice(index, 0, message);
    
    // 更新最后消息序列号
    if (message.seqId > (lastSeqMap.value[chatId] || 0)) {
      lastSeqMap.value[chatId] = message.seqId;
    }
  }
  
  function findInsertIndex(msgList: Message[], seqId: number): number {
    let left = 0;
    let right = msgList.length;
    
    while (left < right) {
      const mid = Math.floor((left + right) / 2);
      if (msgList[mid].seqId < seqId) {
        left = mid + 1;
      } else {
        right = mid;
      }
    }
    
    return left;
  }
  
  function getLastSeq(chatId: string): number {
    return lastSeqMap.value[chatId] || 0;
  }
  
  function removeMessage(chatId: string, msgId: string) {
    if (messages.value[chatId]) {
      const index = messages.value[chatId].findIndex(m => m.id === msgId);
      if (index !== -1) {
        messages.value[chatId].splice(index, 1);
      }
    }
  }
  
  function updateMessage(chatId: string, msgId: string, updates: Partial<Message>) {
    if (messages.value[chatId]) {
      const message = messages.value[chatId].find(m => m.id === msgId);
      if (message) {
        Object.assign(message, updates);
      }
    }
  }
  
  function loadHistory(chatId: string, historyMessages: Message[]) {
    messages.value[chatId] = historyMessages;
    if (historyMessages.length > 0) {
      const maxSeq = Math.max(...historyMessages.map(m => m.seqId));
      lastSeqMap.value[chatId] = maxSeq;
    }
  }
  
  function addToSendQueue(message: Message) {
    sendQueue.value.push(message);
  }
  
  function removeFromSendQueue(clientMsgId: string) {
    const index = sendQueue.value.findIndex(m => m.clientMsgId === clientMsgId);
    if (index !== -1) {
      sendQueue.value.splice(index, 1);
    }
  }
  
  return {
    activeChatId,
    messages,
    lastSeqMap,
    sendQueue,
    chatSessions,
    currentMessages,
    unreadCount,
    setActiveChat,
    addMessage,
    getLastSeq,
    removeMessage,
    updateMessage,
    loadHistory,
    addToSendQueue,
    removeFromSendQueue
  };
});
```

#### 3.2.2 User Store设计

```typescript
// stores/user.ts

import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { UserInfo } from '@/types/user';

export const useUserStore = defineStore('user', () => {
  const userInfo = ref<UserInfo | null>(null);
  const token = ref<string>('');
  const onlineUsers = ref<Set<number>>(new Set());
  
  const isLoggedIn = computed(() => !!token.value);
  const userId = computed(() => userInfo.value?.id || 0);
  const userRole = computed(() => userInfo.value?.role || 'STUDENT');
  
  function setUserInfo(info: UserInfo) {
    userInfo.value = info;
  }
  
  function setToken(tok: string) {
    token.value = tok;
    localStorage.setItem('token', tok);
  }
  
  function logout() {
    userInfo.value = null;
    token.value = '';
    localStorage.removeItem('token');
  }
  
  function updateOnlineStatus(userId: number, isOnline: boolean) {
    if (isOnline) {
      onlineUsers.value.add(userId);
    } else {
      onlineUsers.value.delete(userId);
    }
  }
  
  function isUserOnline(userId: number): boolean {
    return onlineUsers.value.has(userId);
  }
  
  // 初始化时从localStorage恢复token
  const savedToken = localStorage.getItem('token');
  if (savedToken) {
    token.value = savedToken;
  }
  
  return {
    userInfo,
    token,
    onlineUsers,
    isLoggedIn,
    userId,
    userRole,
    setUserInfo,
    setToken,
    logout,
    updateOnlineStatus,
    isUserOnline
  };
});
```

```typescript
/**
 * 消息去重
 */
export function deduplicateMessages(messages: Message[]): Message[] {
  const seen = new Set<string>();
  return messages.filter(msg => {
    if (seen.has(msg.clientMsgId || msg.id)) {
      return false;
    }
    seen.add(msg.clientMsgId || msg.id);
    return true;
  });
}

/**
 * 二分法插入排序
 */
export function insertMessageSorted(messages: Message[], newMessage: Message): void {
  const index = findInsertIndex(messages, newMessage.seqId);
  messages.splice(index, 0, newMessage);
}

function findInsertIndex(messages: Message[], seqId: number): number {
  let left = 0;
  let right = messages.length;
  
  while (left < right) {
    const mid = Math.floor((left + right) / 2);
    if (messages[mid].seqId < seqId) {
      left = mid + 1;
    } else {
      right = mid;
    }
  }
  
  return left;
}
```

### 5.1 虚拟滚动优化

```typescript
// composables/useVirtualScroll.ts

import { ref, computed } from 'vue';

export function useVirtualScroll(itemHeight: number, visibleCount: number) {
  const scrollTop = ref(0);
  const totalItems = ref(0);
  
  const startIndex = computed(() => {
    return Math.floor(scrollTop.value / itemHeight);
  });
  
  const endIndex = computed(() => {
    return Math.min(startIndex.value + visibleCount, totalItems.value);
  });
  
  const visibleItems = computed(() => {
    return {
      start: startIndex.value,
      end: endIndex.value
    };
  });
  
  const containerHeight = computed(() => {
    return totalItems.value * itemHeight;
  });
  
  function handleScroll(e: Event) {
    const target = e.target as HTMLElement;
    scrollTop.value = target.scrollTop;
  }
  
  return {
    scrollTop,
    totalItems,
    visibleItems,
    containerHeight,
    handleScroll
  };
}
```

```typescript
// utils/debounce.ts

export function debounce<T extends (...args: any[]) => any>(
  fn: T,
  delay: number
): (...args: Parameters<T>) => void {
  let timer: number | null = null;
  
  return function(...args: Parameters<T>) {
    if (timer) {
      clearTimeout(timer);
    }
    timer = window.setTimeout(() => {
      fn(...args);
    }, delay);
  };
}

export function throttle<T extends (...args: any[]) => any>(
  fn: T,
  interval: number
): (...args: Parameters<T>) => void {
  let lastTime = 0;
  
  return function(...args: Parameters<T>) {
    const now = Date.now();
    if (now - lastTime >= interval) {
      fn(...args);
      lastTime = now;
    }
  };
}
```
