### 4.2 WebSocket管理器（WebSocketManager）

**核心功能**：
- 连接管理（建立、断开、重连）
- 心跳机制
- 消息队列（去重、排序）
- 消息分发

**设计实现**：
```typescript
class WebSocketManager {
  private connection: WebSocket | null = null;
  private reconnectCount = 0;
  private maxReconnectCount = 5;
  private messageQueue: Message[] = [];
  private lastSeqMap: Record<string, number> = {};
  
  // 连接
  connect(): void {
    this.connection = new WebSocket(this.url);
    this.connection.onopen = this.onOpen.bind(this);
    this.connection.onmessage = this.onMessage.bind(this);
    this.connection.onerror = this.onError.bind(this);
    this.connection.onclose = this.onClose.bind(this);
  }
  
  // 断线重连（指数退避）
  private attemptReconnect(): void {
    if (this.reconnectCount >= this.maxReconnectCount) return;
    this.reconnectCount++;
    const delay = this.reconnectInterval * Math.pow(2, this.reconnectCount - 1);
    setTimeout(() => this.connect(), delay);
  }
  
  // 发送消息
  send(message: Message): void {
    if (this.connection?.readyState === WebSocket.OPEN) {
      this.connection.send(JSON.stringify(message));
    } else {
      this.messageQueue.push(message);
    }
  }
  
  // 消息去重与排序
  private processMessage(message: Message): void {
    const chatId = message.chatId;
    const seqId = message.seqId;
    
    // 去重
    if (this.lastSeqMap[chatId] >= seqId) return;
    
    // 更新序号
    this.lastSeqMap[chatId] = seqId;
    
    // 分发消息
    this.emit('message', message);
  }
}
```

#### 4.3.2 聊天状态（chatStore）

```typescript
export const useChatStore = defineStore('chat', {
  state: () => ({
    activeChatId: '',
    messages: {} as Record<string, MessageVO[]>,
    lastSeqMap: {} as Record<string, number>,
    sendQueue: [] as MessageVO[],
    unreadCounts: {} as Record<string, number>,
  }),
  
  actions: {
    setActiveChat(chatId: string) {
      this.activeChatId = chatId;
      this.resetUnread(chatId);
    },
    
    addMessage(message: MessageVO) {
      const chatId = message.chatId;
      if (!this.messages[chatId]) {
        this.messages[chatId] = [];
      }
      
      // 二分法插入，保持有序
      const index = this.messages[chatId].findIndex(
        m => m.seqId > message.seqId
      );
      if (index === -1) {
        this.messages[chatId].push(message);
      } else {
        this.messages[chatId].splice(index, 0, message);
      }
      
      // 更新最后序号
      this.lastSeqMap[chatId] = Math.max(
        this.lastSeqMap[chatId] || 0,
        message.seqId
      );
      
      // 更新未读计数
      if (chatId !== this.activeChatId) {
        this.unreadCounts[chatId] = (this.unreadCounts[chatId] || 0) + 1;
      }
    },
    
    resetUnread(chatId: string) {
      this.unreadCounts[chatId] = 0;
    },
  },
});
```

router.beforeEach((to, from, next) => {
  const userStore = useUserStore();
  const isLoggedIn = userStore.isLoggedIn;
  
  // 未登录访问需要认证的页面
  if (to.meta.requiresAuth && !isLoggedIn) {
    next('/login');
    return;
  }
  
  // 已登录访问登录/注册页
  if ((to.path === '/login' || to.path === '/register') && isLoggedIn) {
    next('/');
    return;
  }
  
  // 管理员权限校验
  if (to.meta.requiresAdmin && userStore.role !== 2) {
    next('/');
    return;
  }
  
  next();
});

## 6. API接口封装

### 6.1 统一请求封装

```typescript
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';

const api: AxiosInstance = axios.create({
  baseURL: '/api',
  timeout: 10000,
});

// 请求拦截器
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// 响应拦截器
api.interceptors.response.use(
  (response) => response.data,
  (error) => {
    // 处理401未授权
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default api;
```

## 7. WebSocket通信设计

### 7.1 消息协议

**消息格式**：
```typescript
interface WSMessage {
  type: MessageType; // PRIVATE_CHAT | GROUP_CHAT | AI_MESSAGE | SYSTEM
  chatId: string;    // 会话ID
  clientMsgId: string; // 客户端生成的消息ID（去重用）
  content: string;   // 消息内容
  msgType: number;   // 消息类型（文本/图片/文件/AI）
  mentionedUserIds?: number[]; // @提及用户ID列表
  fileUrl?: string;  // 文件URL
  fileName?: string; // 文件名
}
```

### 7.2 消息处理流程

```
客户端发送消息
      │
      ▼
生成clientMsgId（UUID）
      │
      ▼
存入发送队列（等待确认）
      │
      ▼
WebSocket发送
      │
      ▼
服务端响应（返回seqId）
      │
      ▼
从发送队列移除
      │
      ▼
更新本地消息状态
```

const lazyLoad = {
  mounted(el: HTMLElement, binding: DirectiveBinding) {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            el.setAttribute('src', binding.value);
            observer.unobserve(el);
          }
        });
      },
      { rootMargin: '100px' }
