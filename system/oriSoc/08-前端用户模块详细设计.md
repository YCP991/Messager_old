## 1. 模块概述

### 1.1 模块定位
用户模块负责用户信息管理、好友关系维护、在线状态同步等核心功能，是系统身份认证和社交关系的基础模块。

### 1.2 核心功能
- 用户信息管理（头像、昵称、签名等）
- 好友关系管理（添加、删除、分组）
- 在线状态同步
- 用户搜索
- 好友排序与置顶

```vue
<script setup lang="ts">
import { computed } from 'vue';

  showStatus?: boolean;
  showName?: boolean;

  size: 'medium',
  showStatus: true,
  showName: true,

  click: [user: UserVO];
```

### 2.2 好友列表组件（FriendList）

#### 2.2.1 功能需求
- 展示好友列表
- 支持分组显示
- 支持搜索过滤
- 支持拖拽排序
- 显示在线状态

#### 2.2.2 Props设计

| Prop | 类型 | 必填 | 说明 |
|------|------|------|------|
| friends | FriendVO[] | 是 | 好友列表 |
| groups | FriendGroup[] | 否 | 分组列表 |
| searchQuery | string | 否 | 搜索关键词 |

#### 2.2.3 Events设计

| Event | 参数 | 说明 |
|-------|------|------|
| select | friend: FriendVO | 选中好友 |
| sort | friends: FriendVO[] | 排序变更 |

#### 2.2.4 组件结构

```vue
<template>
  <div class="friend-list">
    <!-- 分组列表 -->
    <div v-for="group in displayGroups" :key="group.id" class="friend-group">
      <div class="group-header">
        <span class="group-name">{{ group.name }}</span>
        <span class="group-count">{{ getGroupFriends(group.id).length }}</span>
      </div>
      
      <div class="friend-items">
        <div
          v-for="friend in getGroupFriends(group.id)"
          :key="friend.id"
          class="friend-item"
          :class="{ active: selectedId === friend.id }"
          draggable="true"
          @dragstart="handleDragStart($event, friend)"
          @dragover.prevent
          @drop="handleDrop($event, friend)"
          @click="$emit('select', friend)"
        >
          <UserInfo :user="friend" size="small" />
          <div class="friend-extra">
            <span v-if="friend.remark" class="remark">{{ friend.remark }}</span>
            <span class="class-name">{{ friend.classId }}</span>
          </div>
        </div>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue';
import UserInfo from './UserInfo.vue';
import type { FriendVO, FriendGroup } from '@/types';

const props = defineProps<{
  friends: FriendVO[];
  groups?: FriendGroup[];
  searchQuery?: string;
}>();

const emit = defineEmits<{
  select: [friend: FriendVO];
  sort: [friends: FriendVO[]];
}>();

const selectedId = ref<number | null>(null);
const draggingFriend = ref<FriendVO | null>(null);

const displayGroups = computed(() => {
  return props.groups || [
    { id: 0, name: '好友', sortOrder: 0 },
    { id: 1, name: '同学', sortOrder: 1 },
    { id: 2, name: '老师', sortOrder: 2 },
  ];
});

const filteredFriends = computed(() => {
  if (!props.searchQuery) return props.friends;
  
  const query = props.searchQuery.toLowerCase();
  return props.friends.filter(f => 
    f.realName.toLowerCase().includes(query) ||
    f.username.toLowerCase().includes(query) ||
    f.remark?.toLowerCase().includes(query)
  );
});

function getGroupFriends(groupId: number): FriendVO[] {
  return filteredFriends.value
    .filter(f => f.groupId === groupId)
    .sort((a, b) => (a.sortOrder || 0) - (b.sortOrder || 0));
}

function handleDragStart(event: DragEvent, friend: FriendVO) {
  draggingFriend.value = friend;
  if (event.dataTransfer) {
    event.dataTransfer.effectAllowed = 'move';
  }
}

function handleDrop(event: DragEvent, targetFriend: FriendVO) {
  if (!draggingFriend.value || draggingFriend.value.id === targetFriend.id) return;
  
  const newFriends = [...props.friends];
  const sourceIndex = newFriends.findIndex(f => f.id === draggingFriend.value!.id);
  const targetIndex = newFriends.findIndex(f => f.id === targetFriend.id);
  
  if (sourceIndex !== -1 && targetIndex !== -1) {
    const [removed] = newFriends.splice(sourceIndex, 1);
    newFriends.splice(targetIndex, 0, removed);
    
    // 更新排序序号
    newFriends.forEach((f, index) => {
      f.sortOrder = index;
    });
    
    emit('sort', newFriends);
  }
  
  draggingFriend.value = null;
}
</script>
```

## 3. 状态管理设计

### 3.1 用户状态（userStore）

```typescript
export const useUserStore = defineStore('user', {
  state: () => ({
    // 当前用户信息
    currentUser: null as UserVO | null,
    
    // 用户列表（缓存）
    users: {} as Record<number, UserVO>,
    
    // 好友列表
    friends: [] as FriendVO[],
    
    // 好友分组
    friendGroups: [] as FriendGroup[],
    
    // 在线用户ID列表
    onlineUserIds: new Set<number>(),
  }),
  
  getters: {
    // 在线好友列表
    onlineFriends: (state) => {
      return state.friends.filter(f => state.onlineUserIds.has(f.id));
    },
    
    // 按分组统计的好友数量
    friendCountByGroup: (state) => {
      const count: Record<number, number> = {};
      state.friends.forEach(friend => {
        count[friend.groupId] = (count[friend.groupId] || 0) + 1;
      });
      return count;
    },
  },
  
  actions: {
    // 设置当前用户
    setCurrentUser(user: UserVO) {
      this.currentUser = user;
      this.users[user.id] = user;
    },
    
    // 更新用户信息
    updateUser(user: Partial<UserVO>) {
      if (this.currentUser && user.id === this.currentUser.id) {
        this.currentUser = { ...this.currentUser, ...user };
      }
      if (this.users[user.id!]) {
        this.users[user.id!] = { ...this.users[user.id!], ...user };
      }
    },
    
    // 添加用户到缓存
    addUser(user: UserVO) {
      this.users[user.id] = user;
    },
    
    // 获取用户信息
    getUser(userId: number): UserVO | undefined {
      return this.users[userId];
    },
    
    // 设置好友列表
    setFriends(friends: FriendVO[]) {
      this.friends = friends;
      // 更新用户缓存
      friends.forEach(friend => {
        this.users[friend.id] = { ...friend } as UserVO;
      });
    },
    
    // 添加好友
    addFriend(friend: FriendVO) {
      this.friends.push(friend);
      this.users[friend.id] = { ...friend } as UserVO;
    },
    
    // 删除好友
    removeFriend(friendId: number) {
      this.friends = this.friends.filter(f => f.id !== friendId);
    },
    
    // 更新好友备注
    updateFriendRemark(friendId: number, remark: string) {
      const friend = this.friends.find(f => f.id === friendId);
      if (friend) {
        friend.remark = remark;
      }
    },
    
    // 设置好友分组
    updateFriendGroup(friendId: number, groupId: number) {
      const friend = this.friends.find(f => f.id === friendId);
      if (friend) {
        friend.groupId = groupId;
      }
    },
    
    // 更新好友排序
    updateFriendSort(friends: FriendVO[]) {
      this.friends = friends;
    },
    
    // 设置在线用户
    setOnlineUsers(userIds: number[]) {
      this.onlineUserIds = new Set(userIds);
    },
    
    // 添加在线用户
    addOnlineUser(userId: number) {
      this.onlineUserIds.add(userId);
    },
    
    // 移除在线用户
    removeOnlineUser(userId: number) {
      this.onlineUserIds.delete(userId);
    },
    
    // 切换在线状态
    toggleOnlineStatus(userId: number, online: boolean) {
      if (online) {
        this.onlineUserIds.add(userId);
      } else {
        this.onlineUserIds.delete(userId);
      }
    },
    
    // 设置好友分组列表
    setFriendGroups(groups: FriendGroup[]) {
      this.friendGroups = groups;
    },
    
    // 添加好友分组
    addFriendGroup(group: FriendGroup) {
      this.friendGroups.push(group);
    },
    
    // 删除好友分组
    removeFriendGroup(groupId: number) {
      this.friendGroups = this.friendGroups.filter(g => g.id !== groupId);
      // 将该分组的好友移到默认分组
      this.friends.forEach(friend => {
        if (friend.groupId === groupId) {
          friend.groupId = 0; // 默认分组
        }
      });
    },
    
    // 更新分组名称
    updateFriendGroupName(groupId: number, name: string) {
      const group = this.friendGroups.find(g => g.id === groupId);
      if (group) {
        group.name = name;
      }
    },
  },
});
```

## 4. 在线状态同步机制

### 4.1 状态同步流程

```
WebSocket连接建立
      │
      ▼
发送当前用户在线状态
      │
      ▼
服务端广播在线状态
      │
      ▼
客户端接收在线状态更新
      │
      ▼
更新本地在线用户集合
      │
      ▼
触发UI重新渲染
```

### 4.2 在线状态管理实现

```typescript
// WebSocket消息处理器
function handleOnlineStatus(message: OnlineStatusMessage) {
  const { userId, online } = message;
  
  // 更新在线用户集合
  if (online) {
    userStore.addOnlineUser(userId);
  } else {
    userStore.removeOnlineUser(userId);
  }
  
  // 更新用户缓存中的在线状态
  const user = userStore.getUser(userId);
  if (user) {
    user.online = online;
    userStore.addUser({ ...user });
  }
}

// 页面加载时获取在线好友列表
async function loadOnlineFriends() {
  try {
    const response = await api.get('/api/user/online');
    userStore.setOnlineUsers(response.data.userIds);
  } catch (error) {
    console.error('获取在线用户列表失败:', error);
  }
}

// 监听WebSocket连接状态
function handleWebSocketStatus(status: 'connected' | 'disconnected') {
  if (status === 'connected') {
    // 连接建立，发送在线状态
    sendOnlineStatus(true);
  } else {
    // 连接断开，发送离线状态
    sendOnlineStatus(false);
  }
}

function sendOnlineStatus(online: boolean) {
  if (!userStore.currentUser) return;
  
  websocketManager.send({
    type: 'ONLINE_STATUS',
    userId: userStore.currentUser.id,
    online,
    timestamp: Date.now(),
  });
}
```

### 5.2 添加好友实现

```typescript
async function addFriend(userId: number) {
  try {
    const response = await api.post('/api/friend/add', {
      friendUserId: userId,
      message: '您好，我想添加您为好友',
    });
    
    if (response.code === 200) {
      showToast('好友请求已发送');
    }
  } catch (error) {
    console.error('添加好友失败:', error);
    showToast('添加好友失败');
  }
}

async function handleFriendRequest(request: FriendRequest) {
  if (request.status === 'pending') {
    // 显示好友请求提示
    showNotification({
      type: 'friend-request',
      data: request,
    });
  }
}

async function acceptFriendRequest(requestId: number) {
  try {
    await api.post('/api/friend/accept', { requestId });
    showToast('已添加好友');
  } catch (error) {
    console.error('接受好友请求失败:', error);
  }
}

async function rejectFriendRequest(requestId: number) {
  try {
    await api.post('/api/friend/reject', { requestId });
    showToast('已拒绝好友请求');
  } catch (error) {
    console.error('拒绝好友请求失败:', error);
  }
}
```
