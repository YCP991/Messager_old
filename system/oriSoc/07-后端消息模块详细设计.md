## 1. 模块概述

### 1.1 模块定位
消息模块是后端核心模块之一，负责消息的发送、存储、缓存、路由和推送，是实现即时通讯功能的关键组件。

### 1.2 核心功能
- 消息发送（私聊、群聊）
- 消息持久化
- 消息缓存管理
- 消息路由与推送
- 历史消息查询
- 消息同步（断线重连）
- 消息撤回

```java
     */
    private Integer msgType;
    
    /**
     * @提及用户ID列表（JSON数组）
     */
    private String mentionedUserIds;
    
    /**
     * 文件URL
     */
    private String fileUrl;
    
    /**
     * 文件名
     */
    private String fileName;
    
    /**
     * 文件大小
     */
    private Long fileSize;
    
    /**
     * 是否撤回
     */
    @TableField("is_revoked")
    private Boolean revoked;
    
    /**
     * 被替换的消息ID（AI占位替换用）
     */
    @TableField("replace_msg_id")
    private Long replaceMsgId;
    
    /**
     * 创建时间
     */
    @TableField(value = "create_time", fill = FieldFill.INSERT)
    private Date createTime;
    
    /**
     * 转换为VO
     */
    public MessageVO toVO() {
        return MessageVO.builder()
            .id(this.id)
            .chatId(this.chatId)
            .seqId(this.seqId)
            .fromUid(this.fromUid)
            .content(this.content)
            .msgType(this.msgType)
            .mentionedUserIds(parseMentionedIds(this.mentionedUserIds))
            .fileUrl(this.fileUrl)
            .fileName(this.fileName)
            .fileSize(this.fileSize)
            .isRevoked(this.revoked)
            .createTime(this.createTime.toString())
            .build();
    }
    
    private List<Long> parseMentionedIds(String json) {
        if (StringUtils.isEmpty(json)) {
            return Collections.emptyList();
        }
        return JSON.parseArray(json, Long.class);
    }
}
```

### 3.3 MessageServiceImpl实现类

```java
@Service
public class MessageServiceImpl implements MessageService {
    
    @Autowired
    private MessageMapper messageMapper;
    
    @Autowired
    private RedisUtil redisUtil;
    
    @Autowired
    private SnowflakeIdGenerator snowflakeIdGenerator;
    
    @Autowired
    private UserSessionManager userSessionManager;
    
    @Autowired
    private MessageCacheManager messageCacheManager;
    
    @Autowired
    private AsyncTaskExecutor messageExecutor;
    
    @Override
    public MessageVO sendMessage(SendMessageDTO dto) {
        // 1. 消息去重检查
        String dedupKey = RedisKeyConstants.CLIENT_MSG_PREFIX + dto.getClientMsgId();
        Boolean exists = redisUtil.setnx(dedupKey, "1", 300); // 5分钟过期
        if (Boolean.TRUE.equals(exists)) {
            // 已存在，说明重复发送
            throw new BusinessException("消息重复");
        }
        
        // 2. 双轴定序
        Long globalId = snowflakeIdGenerator.nextId();
        Long seqId = redisUtil.incr(RedisKeyConstants.SEQ_ID_PREFIX + dto.getChatId());
        
        // 3. 构建消息实体
        Message message = Message.builder()
            .id(globalId)
            .chatId(dto.getChatId())
            .seqId(seqId)
            .fromUid(dto.getFromUid())
            .content(dto.getContent())
            .msgType(dto.getMsgType())
            .mentionedUserIds(formatMentionedIds(dto.getMentionedUserIds()))
            .fileUrl(dto.getFileUrl())
            .fileName(dto.getFileName())
            .fileSize(dto.getFileSize())
            .revoked(false)
            .build();
        
        // 4. 异步持久化
        messageExecutor.execute(() -> {
            messageMapper.insert(message);
        });
        
        // 5. 更新缓存
        messageCacheManager.addMessage(message);
        
        // 6. 发布到消息总线
        redisUtil.xAdd(RedisKeyConstants.STREAM_MESSAGE, message.toJson());
        
        // 7. 本地推送
        pushMessage(message);
        
        return message.toVO();
    }
    
    @Override
    public List<MessageVO> getHistoryMessages(String chatId, Long lastSeq, Integer size) {
        // 优先从缓存获取
        List<Message> cacheMessages = messageCacheManager.getMessages(chatId, lastSeq, size);
        
        if (cacheMessages.size() >= size) {
            return cacheMessages.stream().map(Message::toVO).collect(Collectors.toList());
        }
        
        // 缓存不足，从数据库补充
        List<Message> dbMessages = messageMapper.selectHistory(chatId, lastSeq, size - cacheMessages.size());
        
        List<Message> result = new ArrayList<>();
        result.addAll(dbMessages);
        result.addAll(cacheMessages);
        
        return result.stream().map(Message::toVO).collect(Collectors.toList());
    }
    
    @Override
    public List<MessageVO> syncMessages(String chatId, Long lastSeq) {
        // 先从缓存获取增量
        List<Message> cacheMessages = messageCacheManager.getMessagesAfterSeq(chatId, lastSeq);
        
        if (!cacheMessages.isEmpty()) {
            Long maxCacheSeq = cacheMessages.stream()
                .map(Message::getSeqId)
                .max(Long::compareTo)
                .orElse(lastSeq);
            
            // 缓存覆盖了所有增量
            if (maxCacheSeq > lastSeq) {
                return cacheMessages.stream().map(Message::toVO).collect(Collectors.toList());
            }
        }
        
        // 从数据库获取
        List<Message> dbMessages = messageMapper.selectAfterSeq(chatId, lastSeq);
        
        // 合并缓存和数据库数据
        Set<Long> cacheIds = cacheMessages.stream()
            .map(Message::getId)
            .collect(Collectors.toSet());
        
        List<Message> merged = new ArrayList<>();
        merged.addAll(dbMessages.stream()
            .filter(m -> !cacheIds.contains(m.getId()))
            .collect(Collectors.toList()));
        merged.addAll(cacheMessages);
        
        // 按seqId排序
        merged.sort(Comparator.comparingLong(Message::getSeqId));
        
        return merged.stream().map(Message::toVO).collect(Collectors.toList());
    }
    
    @Override
    public boolean revokeMessage(Long messageId, String chatId, Long userId) {
        Message message = messageMapper.selectById(messageId);
        
        if (message == null) {
            throw new BusinessException("消息不存在");
        }
        
        // 校验权限（只有发送者或管理员可撤回）
        if (!message.getFromUid().equals(userId)) {
            // 检查是否为群管理员
            boolean isAdmin = checkGroupAdmin(chatId, userId);
            if (!isAdmin) {
                throw new BusinessException("无权限撤回消息");
            }
        }
        
        // 检查撤回时间限制（2分钟内）
        Date now = new Date();
        long diff = now.getTime() - message.getCreateTime().getTime();
        if (diff > 120000) { // 2分钟
            throw new BusinessException("消息超过撤回时间");
        }
        
        // 更新消息状态
        message.setRevoked(true);
        messageMapper.updateById(message);
        
        // 更新缓存
        messageCacheManager.updateMessage(message);
        
        // 推送撤回通知
        pushRevokeMessage(message);
        
        return true;
    }
    
    private void pushMessage(Message message) {
        String chatId = message.getChatId();
        String json = message.toJson();
        
        if (chatId.startsWith("p_")) {
            // 私聊：推送给对方
            Long targetUserId = extractTargetUserId(chatId);
            userSessionManager.pushToUser(targetUserId, json);
        } else if (chatId.startsWith("g_")) {
            // 群聊：推送给所有在线成员
            Long groupId = extractGroupId(chatId);
            List<Long> onlineMembers = getOnlineMembers(groupId);
            onlineMembers.forEach(userId -> 
                userSessionManager.pushToUser(userId, json)
            );
        }
    }
    
    private void pushRevokeMessage(Message message) {
        // 构建撤回通知消息
        Map<String, Object> revokeMsg = new HashMap<>();
        revokeMsg.put("type", "REVOKE");
        revokeMsg.put("messageId", message.getId());
        revokeMsg.put("chatId", message.getChatId());
        
        pushMessageToChat(message.getChatId(), JSON.toJSONString(revokeMsg));
    }
    
    private void pushMessageToChat(String chatId, String message) {
        if (chatId.startsWith("p_")) {
            Long targetUserId = extractTargetUserId(chatId);
            userSessionManager.pushToUser(targetUserId, message);
            // 也推送给发送者
            // ...
        } else if (chatId.startsWith("g_")) {
            Long groupId = extractGroupId(chatId);
            List<Long> onlineMembers = getOnlineMembers(groupId);
            onlineMembers.forEach(userId -> 
                userSessionManager.pushToUser(userId, message)
            );
        }
    }
    
    private Long extractTargetUserId(String chatId) {
        // chatId格式：p_uid1_uid2
        String[] parts = chatId.substring(2).split("_");
        // 返回不是当前用户的那个ID
        return Long.parseLong(parts[0]);
    }
    
    private Long extractGroupId(String chatId) {
        // chatId格式：g_gid
        return Long.parseLong(chatId.substring(2));
    }
    
    private String formatMentionedIds(List<Long> ids) {
        if (ids == null || ids.isEmpty()) {
            return null;
        }
        return JSON.toJSONString(ids);
    }
}
```

## 4. 缓存管理设计

### 4.1 MessageCacheManager

```java
@Component
public class MessageCacheManager {
    
    @Autowired
    private RedisUtil redisUtil;
    
    /**
     * 添加消息到缓存
     */
    public void addMessage(Message message) {
        String key = RedisKeyConstants.CHAT_MSG + message.getChatId();
        
        // 使用ZSet存储，score为seqId
        redisUtil.zAdd(key, message.toJson(), message.getSeqId());
        
        // 保留最近100条消息
        redisUtil.zRemRangeByRank(key, 0, -101);
    }
    
    /**
     * 获取消息列表
     */
    public List<Message> getMessages(String chatId, Long lastSeq, Integer size) {
        String key = RedisKeyConstants.CHAT_MSG + chatId;
        
        // 获取score < lastSeq的消息，按score降序排列
        Set<String> messages = redisUtil.zRevRangeByScore(key, 0, lastSeq - 1, 0, size);
        
        return messages.stream()
            .map(json -> JSON.parseObject(json, Message.class))
            .collect(Collectors.toList());
    }
    
    /**
     * 获取指定序号之后的消息
     */
    public List<Message> getMessagesAfterSeq(String chatId, Long seqId) {
        String key = RedisKeyConstants.CHAT_MSG + chatId;
        
        Set<String> messages = redisUtil.zRangeByScore(key, seqId + 1, Double.MAX_VALUE);
        
        return messages.stream()
            .map(json -> JSON.parseObject(json, Message.class))
            .sorted(Comparator.comparingLong(Message::getSeqId))
            .collect(Collectors.toList());
    }
    
    /**
     * 更新消息
     */
    public void updateMessage(Message message) {
        String key = RedisKeyConstants.CHAT_MSG + message.getChatId();
        
        // 先删除旧的
        redisUtil.zRem(key, message.toJson());
        
        // 添加新的
        redisUtil.zAdd(key, message.toJson(), message.getSeqId());
    }
    
    /**
     * 获取缓存的消息数量
     */
    public long getCacheSize(String chatId) {
        String key = RedisKeyConstants.CHAT_MSG + chatId;
        return redisUtil.zCard(key);
    }
    
    /**
     * 清理缓存
     */
    public void clearCache(String chatId) {
        String key = RedisKeyConstants.CHAT_MSG + chatId;
        redisUtil.del(key);
    }
}
```

```java
@Component
public class MessageTableRouter {
    
    /**
     * 根据消息ID获取表名
     */
    public String getTableName(Long messageId) {
        // 雪花ID前41位是时间戳（毫秒）
        long timestamp = messageId >> 22;
        Date date = new Date(timestamp);
        return formatTableName(date);
    }
    
    /**
     * 根据日期获取表名
     */
    public String getTableName(Date date) {
        return formatTableName(date);
    }
    
    /**
     * 获取时间范围内的所有表名
     */
    public List<String> getTableNames(Date startDate, Date endDate) {
        List<String> tableNames = new ArrayList<>();
        Calendar cal = Calendar.getInstance();
        cal.setTime(startDate);
        
        while (!cal.getTime().after(endDate)) {
            tableNames.add(formatTableName(cal.getTime()));
            cal.add(Calendar.MONTH, 1);
        }
        
        return tableNames;
    }
    
    /**
     * 格式化表名
     */
    private String formatTableName(Date date) {
        SimpleDateFormat sdf = new SimpleDateFormat("yyyyMM");
        return "im_message_" + sdf.format(date);
    }
    
    /**
     * 获取当前表名
     */
    public String getCurrentTableName() {
        return formatTableName(new Date());
    }
}
```

### 5.2 MyBatis动态表名拦截器

```java
@Component
@Intercepts({
    @Signature(type = Executor.class, method = "query", args = {MappedStatement.class, Object.class, RowBounds.class, ResultHandler.class}),
    @Signature(type = Executor.class, method = "update", args = {MappedStatement.class, Object.class})
})
public class DynamicTableNameInterceptor implements Interceptor {
    
    @Autowired
    private MessageTableRouter tableRouter;
    
    @Override
    public Object intercept(Invocation invocation) throws Throwable {
        // 获取MappedStatement
        MappedStatement ms = (MappedStatement) invocation.getArgs()[0];
        
        // 判断是否需要动态表名
        if (ms.getId().contains("MessageMapper")) {
            // 获取参数
            Object parameter = invocation.getArgs()[1];
            
            // 根据参数获取表名
            String tableName = determineTableName(ms.getId(), parameter);
            
            // 修改SQL中的表名
            BoundSql boundSql = ms.getBoundSql(parameter);
            String sql = boundSql.getSql();
            String newSql = replaceTableName(sql, tableName);
            
            // 重新设置SQL
            BoundSql newBoundSql = new BoundSql(ms.getConfiguration(), newSql, 
                boundSql.getParameterMappings(), boundSql.getParameterObject());
            
            // 创建新的MappedStatement
            MappedStatement newMs = copyMappedStatement(ms, newBoundSql);
            invocation.getArgs()[0] = newMs;
        }
        
        return invocation.proceed();
    }
    
    private String determineTableName(String mapperId, Object parameter) {
        // 根据不同的方法和参数确定表名
        if (mapperId.endsWith("insert")) {
            return tableRouter.getCurrentTableName();
        } else if (mapperId.endsWith("selectById")) {
            Long id = (Long) parameter;
            return tableRouter.getTableName(id);
        } else if (parameter instanceof Map) {
            Map<String, Object> params = (Map<String, Object>) parameter;
            if (params.containsKey("messageId")) {
                Long id = (Long) params.get("messageId");
                return tableRouter.getTableName(id);
            } else if (params.containsKey("chatId") && params.containsKey("lastSeq")) {
                // 查询历史消息，需要考虑跨表
                return "im_message"; // 使用视图或合并查询
            }
        }
        
        return "im_message";
    }
    
    private String replaceTableName(String sql, String tableName) {
        return sql.replace("im_message", tableName);
    }
    
    private MappedStatement copyMappedStatement(MappedStatement ms, BoundSql boundSql) {
        MappedStatement.Builder builder = new MappedStatement.Builder(
            ms.getConfiguration(), ms.getId(), 
            new BoundSqlSource(boundSql), ms.getSqlCommandType()
        );
        
        builder.resource(ms.getResource())
            .fetchSize(ms.getFetchSize())
            .statementType(ms.getStatementType())
            .keyGenerator(ms.getKeyGenerator())
            .keyProperty(StringUtils.join(ms.getKeyProperties(), ","))
            .timeout(ms.getTimeout())
            .parameterMap(ms.getParameterMap())
            .resultMaps(ms.getResultMaps())
            .resultSetType(ms.getResultSetType())
            .cache(ms.getCache())
            .flushCacheRequired(ms.isFlushCacheRequired())
            .useCache(ms.isUseCache());
        
        return builder.build();
    }
    
    private static class BoundSqlSource implements SqlSource {
        private final BoundSql boundSql;
        
        public BoundSqlSource(BoundSql boundSql) {
            this.boundSql = boundSql;
        }
        
        @Override
        public BoundSql getBoundSql(Object parameterObject) {
            return boundSql;
        }
    }
    
    @Override
    public Object plugin(Object target) {
        return Plugin.wrap(target, this);
    }
    
    @Override
    public void setProperties(Properties properties) {
        // 无需配置
    }
}
```

### 6.2 消息消费者

```java
@Component
public class MessageStreamConsumer {
    
    @Autowired
    private RedisUtil redisUtil;
    
    @Autowired
    private UserSessionManager userSessionManager;
    
    @Autowired
    private MessageMapper messageMapper;
    
    // 消费组名称
    private static final String CONSUMER_GROUP = "maisizhe-message-group";
    // 消费者名称（使用主机名+端口）
    private String consumerName;
    
    @PostConstruct
    public void init() {
        // 获取主机名作为消费者名称
        try {
            consumerName = InetAddress.getLocalHost().getHostName() + "-" + System.currentTimeMillis();
        } catch (UnknownHostException e) {
            consumerName = "consumer-" + System.currentTimeMillis();
        }
        
        // 创建消费组（如果不存在）
        try {
            redisUtil.xGroupCreate(RedisKeyConstants.STREAM_MESSAGE, CONSUMER_GROUP, "0", true);
        } catch (Exception e) {
            // 消费组已存在，忽略
        }
        
        // 启动消费线程
        startConsuming();
    }
    
    private void startConsuming() {
        ExecutorService executor = Executors.newSingleThreadExecutor();
        executor.submit(() -> {
            while (true) {
                try {
                    // 读取消息
                    List<Map<String, Object>> messages = redisUtil.xReadGroup(
                        CONSUMER_GROUP, consumerName,
                        RedisKeyConstants.STREAM_MESSAGE, ">", 10, 5000
                    );
                    
                    for (Map<String, Object> message : messages) {
                        try {
                            processMessage(message);
                            
                            // 确认消息
                            String messageId = (String) message.get("id");
                            redisUtil.xAck(RedisKeyConstants.STREAM_MESSAGE, CONSUMER_GROUP, messageId);
                        } catch (Exception e) {
                            // 处理失败，消息会留在pending list中
                            log.error("处理消息失败: {}", e.getMessage());
                        }
                    }
                } catch (Exception e) {
                    log.error("消费消息异常: {}", e.getMessage());
                    try {
                        Thread.sleep(1000);
                    } catch (InterruptedException ie) {
                        Thread.currentThread().interrupt();
                        break;
                    }
                }
            }
        });
    }
    
    private void processMessage(Map<String, Object> message) {
        String chatId = (String) message.get("chatId");
        
        if (chatId == null) {
            return;
        }
        
        if (chatId.startsWith("p_")) {
            // 私聊：推送给对方
            Long targetUserId = extractTargetUserId(chatId);
            if (userSessionManager.hasSession(targetUserId)) {
                String json = convertToJson(message);
                userSessionManager.pushToUser(targetUserId, json);
            }
        } else if (chatId.startsWith("g_")) {
            // 群聊：推送给所有在线成员
            Long groupId = extractGroupId(chatId);
            List<Long> onlineMembers = getOnlineMembers(groupId);
            
            String json = convertToJson(message);
            for (Long userId : onlineMembers) {
                userSessionManager.pushToUser(userId, json);
            }
        }
    }
    
    private Long extractTargetUserId(String chatId) {
        String[] parts = chatId.substring(2).split("_");
        return Long.parseLong(parts[0]);
    }
    
    private Long extractGroupId(String chatId) {
        return Long.parseLong(chatId.substring(2));
    }
    
    private String convertToJson(Map<String, Object> message) {
        Map<String, Object> result = new HashMap<>();
        result.put("type", "MESSAGE");
        result.put("chatId", message.get("chatId"));
        result.put("messageId", message.get("messageId"));
        result.put("seqId", message.get("seqId"));
        result.put("fromUid", message.get("fromUid"));
        result.put("content", message.get("content"));
        result.put("msgType", message.get("msgType"));
        result.put("createTime", message.get("createTime"));
        
        return JSON.toJSONString(result);
    }
}
```

```java
public interface MessageMapper extends BaseMapper<Message> {
    
    /**
     * 查询历史消息
     * @param chatId 会话ID
     * @param lastSeq 最后消息序号
     * @param size 获取数量
     * @return 消息列表
     */

    List<Message> selectHistory(@Param("chatId") String chatId, 
                                @Param("lastSeq") Long lastSeq, 
                                @Param("size") Integer size);
    
    /**
     * 查询指定序号之后的消息
     * @param chatId 会话ID
     * @param seqId 序号
     * @return 消息列表
     */
    @Select("SELECT * FROM im_message WHERE chat_id = #{chatId} AND seq_id > #{seqId} ORDER BY seq_id ASC")
    List<Message> selectAfterSeq(@Param("chatId") String chatId, @Param("seqId") Long seqId);
    
    /**
     * 查询未读消息数量
     * @param chatId 会话ID
     * @param lastReadSeq 最后已读序号
     * @return 未读数量
     */
    @Select("SELECT COUNT(*) FROM im_message WHERE chat_id = #{chatId} AND seq_id > #{lastReadSeq}")
    long countUnread(@Param("chatId") String chatId, @Param("lastReadSeq") Long lastReadSeq);
    
    /**
     * 删除消息（软删除）
     * @param id 消息ID
     */
    @Update("UPDATE im_message SET is_revoked = 1 WHERE id = #{id}")
    void revokeById(@Param("id") Long id);
    
    /**
     * 跨表查询历史消息
     * @param chatId 会话ID
     * @param lastSeq 最后消息序号
     * @param size 获取数量
     * @param tableNames 表名列表
     * @return 消息列表
     */
    @Select("<script>" +
            "SELECT * FROM (" +
            "<foreach collection='tableNames' item='tableName' separator=' UNION ALL '>" +
            "SELECT * FROM ${tableName} WHERE chat_id = #{chatId} AND seq_id < #{lastSeq}" +
            "</foreach>" +
            ") AS t ORDER BY seq_id DESC LIMIT #{size}" +
            "</script>")
    List<Message> selectHistoryCrossTable(@Param("chatId") String chatId,
                                         @Param("lastSeq") Long lastSeq,
                                         @Param("size") Integer size,
                                         @Param("tableNames") List<String> tableNames);
}
```
