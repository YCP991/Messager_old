## 1. 模块概述

### 1.1 模块定位
消息模块是前端核心模块之一，负责消息的发送、接收、展示和管理，是实现即时通讯功能的关键组件。

### 1.2 核心功能
- 消息发送（文本、图片、文件）
- 消息接收与展示
- @提及功能
- 消息撤回
- 历史消息加载
- 消息已读状态管理
- 消息去重与排序

## 3. WebSocket管理器设计

### 3.1 核心功能

- 消息队列管理（去重、排序）
- 消息分发

### 3.2 类设计

```typescript
class WebSocketManager {
  private connection: WebSocket | null = null;
  private reconnectCount = 0;
  private maxReconnectCount = 5;
  private reconnectInterval = 1000;
  private messageQueue: Message[] = [];
  private lastSeqMap: Record<string, number> = {};
  private pingTimer: number | null = null;
  
  // 连接配置
  private config = {
    url: '/ws/chat',
    pingInterval: 25000, // 25秒心跳
    pingTimeout: 5000,    // 5秒超时
  };
  
  // 连接
  connect(): void {
    // 关闭现有连接
    this.disconnect();
    
    // 创建新连接
    this.connection = new WebSocket(this.config.url);
    
    // 设置事件处理
    this.connection.onopen = this.onOpen.bind(this);
    this.connection.onmessage = this.onMessage.bind(this);
    this.connection.onerror = this.onError.bind(this);
    this.connection.onclose = this.onClose.bind(this);
  }
  
  // 断开连接
  disconnect(): void {
    if (this.connection) {
      this.connection.close();
      this.connection = null;
    }
    if (this.pingTimer) {
      clearInterval(this.pingTimer);
      this.pingTimer = null;
    }
  }
  
  // 发送消息
  send(message: Message): void {
    if (!this.connection || this.connection.readyState !== WebSocket.OPEN) {
      // 加入发送队列
      this.messageQueue.push(message);
      return;
    }
    
    this.connection.send(JSON.stringify(message));
  }
  
  // 处理接收到的消息
  private onMessage(event: MessageEvent): void {
    const message = JSON.parse(event.data) as Message;
    
    // 心跳响应
    if (message.type === 'PONG') {
      return;
    }
    
    // 消息去重
    const chatId = message.chatId;
    const seqId = message.seqId;
    
    if (this.lastSeqMap[chatId] >= seqId) {
      return; // 已处理过
    }
    
    // 更新最后序号
    this.lastSeqMap[chatId] = seqId;
    
    // 分发消息
    this.emit('message', message);
  }
  
  // 断线重连（指数退避）
  private attemptReconnect(): void {
    if (this.reconnectCount >= this.maxReconnectCount) {
      this.emit('reconnectFailed');
      return;
    }
    
    this.reconnectCount++;
    const delay = this.reconnectInterval * Math.pow(2, this.reconnectCount - 1);
    
    setTimeout(() => {
      this.connect();
    }, delay);
  }
  
  // 心跳发送
  private startHeartbeat(): void {
    this.pingTimer = window.setInterval(() => {
      if (this.connection?.readyState === WebSocket.OPEN) {
        this.connection.send(JSON.stringify({ type: 'PING', timestamp: Date.now() }));
      }
    }, this.config.pingInterval);
  }
  
  // 事件触发
  private emit(event: string, ...args: unknown[]): void {
    // 触发自定义事件
    const customEvent = new CustomEvent(event, { detail: args });
    window.dispatchEvent(customEvent);
  }
}
```

## 4. 状态管理设计

### 4.1 聊天状态（chatStore）

```typescript
export const useChatStore = defineStore('chat', {
  state: () => ({
    // 当前活跃会话ID
    activeChatId: '',
    
    // 消息列表（key为chatId）
    messages: {} as Record<string, MessageVO[]>,
    
    // 每个会话的最后消息序号
    lastSeqMap: {} as Record<string, number>,
    
    // 发送队列（等待确认的消息）
    sendQueue: [] as MessageVO[],
    
    // 未读消息计数
    unreadCounts: {} as Record<string, number>,
    
    // 是否正在加载历史消息
    loadingHistory: false,
    
    // 是否有更多历史消息
    hasMoreHistory: {} as Record<string, boolean>,
  }),
  
  getters: {
    // 当前会话消息列表
    currentMessages(): MessageVO[] {
      return this.messages[this.activeChatId] || [];
    },
    
    // 当前会话未读计数
    currentUnreadCount(): number {
      return this.unreadCounts[this.activeChatId] || 0;
    },
  },
  
  actions: {
    // 设置活跃会话
    setActiveChat(chatId: string) {
      this.activeChatId = chatId;
      this.resetUnread(chatId);
    },
    
    // 添加消息
    addMessage(message: MessageVO) {
      const chatId = message.chatId;
      
      if (!this.messages[chatId]) {
        this.messages[chatId] = [];
      }
      
      // 去重检查
      const exists = this.messages[chatId].some(m => m.id === message.id);
      if (exists) return;
      
      // 有序插入（按seqId排序）
      const index = this.messages[chatId].findIndex(
        m => m.seqId > message.seqId
      );
      
      if (index === -1) {
        this.messages[chatId].push(message);
      } else {
        this.messages[chatId].splice(index, 0, message);
      }
      
      // 更新最后序号
      this.lastSeqMap[chatId] = Math.max(
        this.lastSeqMap[chatId] || 0,
        message.seqId
      );
      
      // 更新未读计数（非当前会话）
      if (chatId !== this.activeChatId) {
        this.unreadCounts[chatId] = (this.unreadCounts[chatId] || 0) + 1;
      }
    },
    
    // 添加多条消息
    addMessages(messages: MessageVO[], chatId: string) {
      messages.forEach(msg => this.addMessage({ ...msg, chatId }));
    },
    
    // 重置未读计数
    resetUnread(chatId: string) {
      this.unreadCounts[chatId] = 0;
    },
    
    // 发送消息（加入发送队列）
    sendMessage(message: MessageVO) {
      this.sendQueue.push(message);
      // 触发WebSocket发送
      websocketManager.send(message);
    },
    
    // 消息发送确认
    confirmMessage(clientMsgId: string) {
      const index = this.sendQueue.findIndex(m => m.clientMsgId === clientMsgId);
      if (index !== -1) {
        this.sendQueue.splice(index, 1);
      }
    },
    
    // 标记消息撤回
    revokeMessage(messageId: number, chatId: string) {
      const messages = this.messages[chatId];
      if (messages) {
        const msg = messages.find(m => m.id === messageId);
        if (msg) {
          msg.isRevoked = true;
        }
      }
    },
    
    // 设置历史消息加载状态
    setLoadingHistory(chatId: string, loading: boolean) {
      this.loadingHistory = loading;
    },
    
    // 设置是否有更多历史消息
    setHasMoreHistory(chatId: string, hasMore: boolean) {
      this.hasMoreHistory[chatId] = hasMore;
    },
  },
});
```

## 5. 消息发送流程

### 5.1 发送流程

```
用户输入消息
      │
      ▼
生成clientMsgId（UUID）
      │
      ▼
构建消息对象
      │
      ▼
加入发送队列（状态：发送中）
      │
      ▼
WebSocket发送
      │
      ▼
服务端响应（返回seqId）
      │
      ▼
从发送队列移除
      │
      ▼
更新消息状态（已发送）
      │
      ▼
触发UI更新
```

### 5.2 发送实现

```typescript
// 在消息输入组件中
async function handleSend() {
  if (!inputValue.value.trim()) return;
  
  // 生成客户端消息ID
  const clientMsgId = uuid();
  
  // 构建消息对象
  const message: MessageVO = {
    id: 0, // 服务端生成
    chatId: props.chatId,
    seqId: 0, // 服务端生成
    fromUid: userStore.id,
    fromUsername: userStore.username,
    fromRealName: userStore.realName,
    content: inputValue.value,
    msgType: 0, // 文本消息
    clientMsgId,
    status: 'sending',
    createTime: new Date().toISOString(),
  };
  
  // 加入发送队列
  chatStore.sendMessage(message);
  
  // 清空输入框
  inputValue.value = '';
}

// WebSocket接收到服务端响应后的处理
function handleMessageResponse(message: MessageVO) {
  // 查找发送队列中的消息
  const pendingMsg = chatStore.sendQueue.find(
    m => m.clientMsgId === message.clientMsgId
  );
  
  if (pendingMsg) {
    // 更新消息ID和seqId
    pendingMsg.id = message.id;
    pendingMsg.seqId = message.seqId;
    pendingMsg.status = 'sent';
    
    // 确认发送
    chatStore.confirmMessage(message.clientMsgId);
  }
}
```

## 6. 历史消息加载

### 6.1 加载流程

```
用户滚动到顶部
      │
      ▼
触发loadMore事件
      │
      ▼
获取当前会话最后一条消息的seqId
      │
      ▼
调用API获取历史消息（lastSeq之前的消息）
      │
      ▼
将历史消息插入列表头部
      │
      ▼
保持滚动位置
```

### 6.2 实现代码

```typescript
// 在消息列表组件中
async function loadMore() {
  if (chatStore.loadingHistory) return;
  
  const chatId = props.chatId;
  const messages = chatStore.messages[chatId];
  
  // 获取最早的消息序号
  const firstMsg = messages?.[0];
  if (!firstMsg) return;
  
  chatStore.setLoadingHistory(chatId, true);
  
  try {
    const response = await api.get('/api/message/history', {
      params: {
        chatId,
        lastSeq: firstMsg.seqId,
        size: 20,
      },
    });
    
    if (response.data.messages.length > 0) {
      // 插入到列表头部
      chatStore.addMessages(response.data.messages, chatId);
      chatStore.setHasMoreHistory(chatId, response.data.hasMore);
    } else {
      chatStore.setHasMoreHistory(chatId, false);
    }
  } catch (error) {
    console.error('加载历史消息失败:', error);
  } finally {
    chatStore.setLoadingHistory(chatId, false);
  }
}
```

### 7.1 去重策略

```typescript
// 客户端生成唯一ID
function generateClientMsgId(): string {
  return `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
}

// 服务端去重检查
// 在服务端消息处理中
function deduplicateMessage(clientMsgId: string): boolean {
  const key = `client:msg:${clientMsgId}`;
  
  // SETNX：如果key不存在则设置，返回true；否则返回false
  const result = redis.setnx(key, '1', 'EX', 300); // 5分钟过期
  
  return !result; // 返回true表示已存在（需要去重）
}
```

### 7.2 排序策略

```typescript
// 客户端排序（按seqId升序）
function sortMessages(messages: MessageVO[]): MessageVO[] {
  return [...messages].sort((a, b) => a.seqId - b.seqId);
}

// 二分法插入（保持有序）
function insertMessage(messages: MessageVO[], newMsg: MessageVO): MessageVO[] {
  const index = messages.findIndex(m => m.seqId > newMsg.seqId);
  
  if (index === -1) {
    messages.push(newMsg);
  } else {
    messages.splice(index, 0, newMsg);
  }
  
  return messages;
}
```
