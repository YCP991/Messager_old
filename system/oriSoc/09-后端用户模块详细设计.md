## 1. 模块概述

### 1.1 模块定位
用户模块是后端核心模块之一，负责用户身份认证、用户信息管理、好友关系维护和在线状态管理，是系统安全和社交功能的基础。

### 1.2 核心功能
- 用户注册与登录
- 用户信息管理
- 好友关系管理（添加、删除、分组）
- 在线状态同步
- 用户搜索

### 3.4 AuthServiceImpl实现类

```java
@Service
public class AuthServiceImpl implements AuthService {
    
    @Autowired
    private UserMapper userMapper;
    
    @Autowired
    private JwtUtil jwtUtil;
    
    @Autowired
    private RedisUtil redisUtil;
    
    @Override
    public Map<String, Object> login(LoginDTO dto) {
        // 1. 查询用户
        User user = userMapper.selectByUsername(dto.getUsername());
        if (user == null) {
            throw new BusinessException("用户不存在");
        }
        
        // 2. 验证密码
        if (!BCrypt.checkpw(dto.getPassword(), user.getPassword())) {
            throw new BusinessException("密码错误");
        }
        
        // 3. 检查用户状态
        if (user.getStatus() != 0) {
            throw new BusinessException("用户已被禁用");
        }
        
        // 4. 生成Token
        String token = jwtUtil.generateToken(user.getId().toString());
        String refreshToken = jwtUtil.generateRefreshToken(user.getId().toString());
        
        // 5. 存储Token到Redis
        String tokenKey = RedisKeyConstants.USER_TOKEN + user.getId();
        redisUtil.set(tokenKey, token, 86400); // 24小时
        
        // 6. 返回结果
        Map<String, Object> result = new HashMap<>();
        result.put("token", token);
        result.put("refreshToken", refreshToken);
        result.put("userInfo", user.toVO());
        
        return result;
    }
    
    @Override
    public UserVO register(RegisterDTO dto) {
        // 1. 检查用户名是否已存在
        User existing = userMapper.selectByUsername(dto.getUsername());
        if (existing != null) {
            throw new BusinessException("用户名已存在");
        }
        
        // 2. 生成用户ID
        Long userId = snowflakeIdGenerator.nextId();
        
        // 3. 密码加密
        String encodedPassword = BCrypt.hashpw(dto.getPassword(), BCrypt.gensalt());
        
        // 4. 创建用户实体
        User user = User.builder()
            .id(userId)
            .username(dto.getUsername())
            .password(encodedPassword)
            .realName(dto.getRealName())
            .role(dto.getRole())
            .classId(dto.getClassId())
            .status(0)
            .build();
        
        // 5. 保存用户
        userMapper.insert(user);
        
        return user.toVO();
    }
    
    @Override
    public Map<String, Object> refreshToken(String refreshToken) {
        // 1. 解析刷新Token
        String userId = jwtUtil.extractUserId(refreshToken);
        if (userId == null) {
            throw new BusinessException("无效的刷新Token");
        }
        
        // 2. 验证用户存在
        User user = userMapper.selectById(Long.parseLong(userId));
        if (user == null) {
            throw new BusinessException("用户不存在");
        }
        
        // 3. 生成新Token
        String token = jwtUtil.generateToken(userId);
        
        // 4. 更新Redis
        String tokenKey = RedisKeyConstants.USER_TOKEN + userId;
        redisUtil.set(tokenKey, token, 86400);
        
        // 5. 返回结果
        Map<String, Object> result = new HashMap<>();
        result.put("token", token);
        
        return result;
    }
    
    @Override
    public Long validateToken(String token) {
        // 1. 解析Token
        String userId = jwtUtil.extractUserId(token);
        if (userId == null) {
            return null;
        }
        
        // 2. 检查Redis中的Token
        String tokenKey = RedisKeyConstants.USER_TOKEN + userId;
        String storedToken = redisUtil.get(tokenKey);
        if (!token.equals(storedToken)) {
            return null;
        }
        
        return Long.parseLong(userId);
    }
}
```

```java
@Service
public class UserServiceImpl implements UserService {
    
    @Autowired
    private UserMapper userMapper;
    
    @Autowired
    private FriendMapper friendMapper;
    
    @Autowired
    private UserSessionManager userSessionManager;
    
    @Override
    public UserVO getUserInfo(Long userId) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BusinessException("用户不存在");
        }
        
        UserVO vo = user.toVO();
        vo.setOnline(userSessionManager.hasSession(userId));
        
        return vo;
    }
    
    @Override
    public UserVO updateUserInfo(Long userId, UpdateUserDTO dto) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BusinessException("用户不存在");
        }
        
        // 更新字段
        if (dto.getRealName() != null) {
            user.setRealName(dto.getRealName());
        }
        if (dto.getPhone() != null) {
            user.setPhone(dto.getPhone());
        }
        if (dto.getEmail() != null) {
            user.setEmail(dto.getEmail());
        }
        if (dto.getSignature() != null) {
            user.setSignature(dto.getSignature());
        }
        
        userMapper.updateById(user);
        
        return user.toVO();
    }
    
    @Override
    public String uploadAvatar(Long userId, MultipartFile file) {
        // 1. 验证文件
        if (file == null || file.isEmpty()) {
            throw new BusinessException("请选择文件");
        }
        
        // 2. 验证文件类型
        String contentType = file.getContentType();
        if (!contentType.contains("image")) {
            throw new BusinessException("只支持图片文件");
        }
        
        // 3. 保存文件
        String fileName = userId + "_" + System.currentTimeMillis() + ".jpg";
        String filePath = "/upload/avatar/" + fileName;
        
        // 实际保存逻辑...
        
        // 4. 更新用户头像
        User user = userMapper.selectById(userId);
        if (user != null) {
            user.setAvatar(filePath);
            userMapper.updateById(user);
        }
        
        return filePath;
    }
    
    @Override
    public void addFriend(Long userId, Long friendUserId, String message) {
        // 1. 检查好友是否已存在
        Friend existing = friendMapper.selectByUserAndFriend(userId, friendUserId);
        if (existing != null) {
            throw new BusinessException("已添加该好友");
        }
        
        // 2. 创建好友关系（双向）
        Friend friend1 = Friend.builder()
            .userId(userId)
            .friendId(friendUserId)
            .groupId(0L) // 默认分组
            .sortOrder(getNextSortOrder(userId))
            .build();
        
        Friend friend2 = Friend.builder()
            .userId(friendUserId)
            .friendId(userId)
            .groupId(0L) // 默认分组
            .sortOrder(getNextSortOrder(friendUserId))
            .build();
        
        friendMapper.insert(friend1);
        friendMapper.insert(friend2);
        
        // 3. 发送好友添加通知
        sendFriendNotification(friendUserId, userId, "ADD_FRIEND", message);
    }
    
    @Override
    public List<FriendVO> getFriendList(Long userId) {
        List<Friend> friends = friendMapper.selectByUserId(userId);
        
        return friends.stream()
            .map(friend -> {
                User user = userMapper.selectById(friend.getFriendId());
                FriendVO vo = new FriendVO();
                vo.setId(user.getId());
                vo.setUsername(user.getUsername());
                vo.setRealName(user.getRealName());
                vo.setRole(user.getRole());
                vo.setClassId(user.getClassId());
                vo.setAvatar(user.getAvatar());
                vo.setRemark(friend.getRemark());
                vo.setGroupId(friend.getGroupId());
                vo.setSortOrder(friend.getSortOrder());
                vo.setOnline(userSessionManager.hasSession(user.getId()));
                return vo;
            })
            .collect(Collectors.toList());
    }
    
    @Override
    public void deleteFriend(Long userId, Long friendId) {
        // 删除双向好友关系
        friendMapper.deleteByUserAndFriend(userId, friendId);
        friendMapper.deleteByUserAndFriend(friendId, userId);
        
        // 发送好友删除通知
        sendFriendNotification(friendId, userId, "DELETE_FRIEND", null);
    }
    
    @Override
    public void updateFriendRemark(Long userId, Long friendId, String remark) {
        Friend friend = friendMapper.selectByUserAndFriend(userId, friendId);
        if (friend == null) {
            throw new BusinessException("好友关系不存在");
        }
        
        friend.setRemark(remark);
        friendMapper.updateById(friend);
    }
    
    @Override
    public List<UserVO> searchUser(String keyword, Integer role) {
        List<User> users = userMapper.searchUser(keyword, role);
        
        return users.stream()
            .map(user -> {
                UserVO vo = user.toVO();
                vo.setOnline(userSessionManager.hasSession(user.getId()));
                return vo;
            })
            .collect(Collectors.toList());
    }
    
    @Override
    public List<Long> getOnlineUserIds() {
        return userSessionManager.getAllOnlineUserIds();
    }
    
    private Double getNextSortOrder(Long userId) {
        Double maxSort = friendMapper.selectMaxSortOrder(userId);
        return (maxSort != null ? maxSort : 0) + 1;
    }
    
    private void sendFriendNotification(Long targetUserId, Long sourceUserId, String type, String message) {
        // 构建通知消息
        Map<String, Object> notification = new HashMap<>();
        notification.put("type", type);
        notification.put("fromUserId", sourceUserId);
        notification.put("message", message);
        
        // 发送通知
        userSessionManager.pushToUser(targetUserId, JSON.toJSONString(notification));
    }
```

## 4. JWT认证过滤器

```java
@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {
    
    @Autowired
    private JwtUtil jwtUtil;
    
    @Autowired
    private RedisUtil redisUtil;
    
    @Autowired
    private UserDetailsService userDetailsService;
    
    @Override
    protected void doFilterInternal(HttpServletRequest request, 
                                    HttpServletResponse response, 
                                    FilterChain filterChain) throws ServletException, IOException {
        // 1. 提取Token
        String token = extractToken(request);
        
        if (token != null) {
            try {
                // 2. 验证Token
                String userId = jwtUtil.extractUserId(token);
                
                if (userId != null && SecurityContextHolder.getContext().getAuthentication() == null) {
                    // 3. 检查Redis中的Token
                    String storedToken = redisUtil.get(RedisKeyConstants.USER_TOKEN + userId);
                    
                    if (token.equals(storedToken)) {
                        // 4. 加载用户信息
                        UserDetails userDetails = userDetailsService.loadUserByUsername(userId);
                        
                        // 5. 设置认证信息
                        UsernamePasswordAuthenticationToken authToken = 
                            new UsernamePasswordAuthenticationToken(
                                userDetails, null, userDetails.getAuthorities()
                            );
                        authToken.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
                        
                        SecurityContextHolder.getContext().setAuthentication(authToken);
                    }
                }
            } catch (Exception e) {
                // Token无效，跳过
            }
        }
        
        filterChain.doFilter(request, response);
    }
    
    private String extractToken(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7);
        }
        return null;
    }
}
```

## 5. 用户会话管理器

```java
@Component
public class UserSessionManager {
    
    /**
     * 用户ID -> WebSocket会话映射
     */
    private final Map<Long, Session> userSessions = new ConcurrentHashMap<>();
    
    /**
     * 添加会话
     */
    public void addSession(Long userId, Session session) {
        userSessions.put(userId, session);
    }
    
    /**
     * 移除会话
     */
    public void removeSession(Long userId) {
        userSessions.remove(userId);
    }
    
    /**
     * 检查用户是否在线
     */
    public boolean hasSession(Long userId) {
        Session session = userSessions.get(userId);
        return session != null && session.isOpen();
    }
    
    /**
     * 获取所有在线用户ID
     */
    public List<Long> getAllOnlineUserIds() {
        return userSessions.entrySet().stream()
            .filter(entry -> entry.getValue().isOpen())
            .map(Map.Entry::getKey)
            .collect(Collectors.toList());
    }
    
    /**
     * 向用户推送消息
     */
    public void pushToUser(Long userId, String message) {
        Session session = userSessions.get(userId);
        
        if (session != null && session.isOpen()) {
            try {
                session.getBasicRemote().sendText(message);
            } catch (IOException e) {
                // 发送失败，移除会话
                removeSession(userId);
            }
        }
    }
    
    /**
     * 批量推送消息
     */
    public void pushToUsers(List<Long> userIds, String message) {
        userIds.forEach(userId -> pushToUser(userId, message));
    }
}
```

```java
    public List<MapRecord<String, String, String>> xReadGroup(String group, String consumer,

        return stringRedisTemplate.opsForStream().read(Consumer.from(group, consumer), 
                                                       StreamReadOptions.empty().count(count).block(block, TimeUnit.MILLISECONDS), 
                                                       offset);

     * Redis Stream确认消息
     */

        stringRedisTemplate.opsForStream().acknowledge(key, group, recordIds);
    }
```
