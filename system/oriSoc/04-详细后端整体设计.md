### 3.2 JWT工具类设计

```java
@Component
public class JwtUtil {
    
    @Value("${jwt.secret}")
    private String secret;
    
    @Value("${jwt.expire}")
    private Long expire;
    
    // 生成Token
    public String generateToken(Long userId, String username) {
        Map<String, Object> claims = new HashMap<>();
        claims.put("userId", userId);
        claims.put("username", username);
        
        return Jwts.builder()
            .setClaims(claims)
            .setSubject(username)
            .setIssuedAt(new Date())
            .setExpiration(new Date(System.currentTimeMillis() + expire * 1000))
            .signWith(SignatureAlgorithm.HS512, secret)
            .compact();
    }
    
    // 解析Token
    public Claims parseToken(String token) {
        return Jwts.parser()
            .setSigningKey(secret)
            .parseClaimsJws(token)
            .getBody();
    }
    
    // 验证Token
    public boolean validateToken(String token) {
        try {
            parseToken(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}
```

## 4. WebSocket网关设计

### 4.1 连接管理

```java
@Component
public class UserSessionManager {
    
    // 用户ID -> WebSocketSession映射
    private final ConcurrentHashMap<Long, WebSocketSession> sessionMap = new ConcurrentHashMap<>();
    
    // 添加会话
    public void addSession(Long userId, WebSocketSession session) {
        sessionMap.put(userId, session);
        // 更新在线状态
        redisUtil.set(RedisKeyConstants.ONLINE_USER + userId, "online", 60);
    }
    
    // 移除会话
    public void removeSession(Long userId) {
        sessionMap.remove(userId);
        redisUtil.del(RedisKeyConstants.ONLINE_USER + userId);
    }
    
    // 获取会话
    public WebSocketSession getSession(Long userId) {
        return sessionMap.get(userId);
    }
    
    // 推送消息
    public void pushToUser(Long userId, String message) {
        WebSocketSession session = sessionMap.get(userId);
        if (session != null && session.isOpen()) {
            session.sendMessage(new TextMessage(message));
        }
    }
}
```

## 5. 消息模块设计

### 5.1 消息发送流程

```
客户端发送消息
      │
      ▼
WebSocket Handler接收
      │
      ▼
消息去重（Redis SETNX）
      │
      ▼
双轴定序（Snowflake ID + Redis INCR）
      │
      ▼
异步持久化（MySQL）
      │
      ▼
热缓存写入（Redis ZSet）
      │
      ▼
消息总线广播（Redis Stream）
      │
      ▼
多节点消费并推送
```

### 5.2 消息服务实现

```java
@Service
public class MessageServiceImpl implements MessageService {
    
    @Autowired
    private MessageMapper messageMapper;
    
    @Autowired
    private RedisUtil redisUtil;
    
    @Autowired
    private SnowflakeIdGenerator snowflakeIdGenerator;
    
    @Autowired
    private UserSessionManager userSessionManager;
    
    @Override
    @Async("messageExecutor")
    public void sendMessage(SendMessageDTO dto) {
        // 1. 生成消息ID
        Long globalId = snowflakeIdGenerator.nextId();
        Long seqId = redisUtil.incr(RedisKeyConstants.SEQ_ID_PREFIX + dto.getChatId());
        
        // 2. 构建消息实体
        Message message = Message.builder()
            .id(globalId)
            .chatId(dto.getChatId())
            .seqId(seqId)
            .fromUid(dto.getFromUid())
            .content(dto.getContent())
            .msgType(dto.getMsgType())
            .build();
        
        // 3. 异步写入数据库
        messageMapper.insert(message);
        
        // 4. 更新缓存
        redisUtil.zAdd(RedisKeyConstants.CHAT_MSG + dto.getChatId(), 
            message.toJson(), seqId);
        
        // 5. 发布到消息总线
        redisUtil.xAdd(RedisKeyConstants.STREAM_MESSAGE, message.toJson());
        
        // 6. 本地推送
        pushMessage(message);
    }
    
    private void pushMessage(Message message) {
        // 根据chatId判断私聊还是群聊
        if (message.getChatId().startsWith("p_")) {
            // 私聊：推送给对方
            Long targetUserId = extractTargetUserId(message.getChatId());
            userSessionManager.pushToUser(targetUserId, message.toJson());
        } else if (message.getChatId().startsWith("g_")) {
            // 群聊：推送给所有在线成员
            Long groupId = extractGroupId(message.getChatId());
            List<Long> onlineMembers = groupMemberService.getOnlineMembers(groupId);
            onlineMembers.forEach(userId -> 
                userSessionManager.pushToUser(userId, message.toJson())
            );
        }
    }
}
```

```java
@Component
public class MessageTableRouter {
    
    // 根据时间获取表名
    public String getTableName(Date date) {
        SimpleDateFormat sdf = new SimpleDateFormat("yyyyMM");
        return "im_message_" + sdf.format(date);
    }
    
    // 根据消息ID获取表名（通过时间戳）
    public String getTableName(Long messageId) {
        // 雪花ID的前41位是时间戳
        long timestamp = messageId >> 22;
        Date date = new Date(timestamp);
        return getTableName(date);
    }
    
    // 跨月查询处理
    public List<String> getTableNames(Date startDate, Date endDate) {
        List<String> tableNames = new ArrayList<>();
        Calendar cal = Calendar.getInstance();
        cal.setTime(startDate);
        
        while (!cal.getTime().after(endDate)) {
            tableNames.add(getTableName(cal.getTime()));
            cal.add(Calendar.MONTH, 1);
        }
        
        return tableNames;
    }
}
```

## 6. AI模块设计

### 6.1 AI异步工作流

```java
@Service
public class AiWorkflowService {
    
    @Autowired
    private MessageService messageService;
    
    @Autowired
    private LlmClient llmClient;
    
    @Autowired
    private MessageCacheManager messageCacheManager;
    
    @Async("aiTaskExecutor")
    public void processAtAiMessage(AtAiTask task) {
        // 1. 立即发送占位消息
        Message placeholder = Message.builder()
            .chatId(task.getChatId())
            .fromUid(0L) // AI用户ID
            .content("AI正在思考中...")
            .msgType(MessageTypeEnum.AI_PLANNING.getCode())
            .build();
        messageService.send(placeholder);
        
        // 2. 拉取上下文（最近5条消息）
        List<Message> context = messageCacheManager.getRecentMessages(task.getChatId(), 5);
        
        // 3. 构建Prompt
        String prompt = buildPrompt(task.getQuestion(), context);
        
        // 4. 调用大模型
        String response = llmClient.call(prompt);
        
        // 5. 发送AI回答
        Message aiResponse = Message.builder()
            .chatId(task.getChatId())
            .fromUid(0L)
            .content(response)
            .msgType(MessageTypeEnum.AI_ANSWER.getCode())
            .replaceMsgId(placeholder.getId()) // 替换占位消息
            .build();
        messageService.send(aiResponse);
    }
    
    private String buildPrompt(String question, List<Message> context) {
        StringBuilder prompt = new StringBuilder();
        prompt.append("你是一个智能助手，请根据上下文回答问题。\n");
        prompt.append("上下文：\n");
        
        for (Message msg : context) {
            prompt.append(msg.getContent()).append("\n");
        }
        
        prompt.append("\n问题：").append(question);
        return prompt.toString();
    }
}
```

### 6.2 定时群日报任务

```java
@Component
public class DailySummaryJob {
    
    @Autowired
    private GroupService groupService;
    
    @Autowired
    private MessageCacheManager messageCacheManager;
    
    @Autowired
    private AiWorkflowService aiWorkflowService;
    
    // 每日21:00执行
    @Scheduled(cron = "0 0 21 * * ?")
    public void generateDailySummary() {
        // 获取所有活跃群组
        List<Group> activeGroups = groupService.getActiveGroups();
        
        for (Group group : activeGroups) {
            // 拉取当日消息
            List<Message> todayMessages = messageCacheManager.getTodayMessages(group.getId());
            
            // 过滤重要消息（基于@次数、回复数加权）
            List<Message> importantMessages = filterImportantMessages(todayMessages);
            
            // 生成摘要任务
            aiWorkflowService.generateSummary(group.getId(), importantMessages);
        }
    }
}
```

```java
public interface GroupCreateStrategy {
    Group create(CreateGroupDTO dto);

// 普通群策略
@Component
public class NormalGroupStrategy implements GroupCreateStrategy {
    @Override
    public Group create(CreateGroupDTO dto) {
        // 直接创建群
        return Group.builder()
            .name(dto.getName())
            .type(GroupTypeEnum.NORMAL.getCode())
            .ownerId(dto.getOwnerId())
            .build();
    }
}

// 班级群策略
@Component
public class ClassGroupStrategy implements GroupCreateStrategy {
    @Override
    public Group create(CreateGroupDTO dto) {
        // 校验用户是否为该班级成员
        validateClassMember(dto.getOwnerId(), dto.getClassId());
        
        Group group = Group.builder()
            .name(dto.getName())
            .type(GroupTypeEnum.CLASS.getCode())
            .ownerId(dto.getOwnerId())
            .build();
        
        // 自动拉取班级成员
        List<Long> classMembers = userService.getClassMembers(dto.getClassId());
        groupMemberService.batchAddMembers(group.getId(), classMembers);
        
        return group;
    }
}

// 工厂类
@Component

    @Autowired
    private Map<String, GroupCreateStrategy> strategyMap;
```

### 9.2 操作日志记录

```java
@Aspect
@Component
public class OperationLogAspect {
    
    @Autowired
    private OperationLogService operationLogService;
    
    @Around("@annotation(com.maisizhe.common.annotation.OperationLog)")
    public Object logOperation(ProceedingJoinPoint joinPoint) throws Throwable {
        long startTime = System.currentTimeMillis();
        
        // 获取操作信息
        MethodSignature signature = (MethodSignature) joinPoint.getSignature();
        OperationLog annotation = signature.getMethod().getAnnotation(OperationLog.class);
        
        // 执行目标方法
        Object result = joinPoint.proceed();
        
        // 记录日志
        OperationLogDTO logDTO = OperationLogDTO.builder()
            .userId(UserContextHolder.getUserId())
            .module(annotation.module())
            .operation(annotation.operation())
            .params(JSON.toJSONString(joinPoint.getArgs()))
            .duration(System.currentTimeMillis() - startTime)
            .build();
        
        operationLogService.save(logDTO);
        
        return result;
    }
}
```
